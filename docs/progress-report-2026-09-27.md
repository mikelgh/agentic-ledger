### 2. 回复草稿生成
```splash
fn generate_reply_draft(email, events) {
    let draft = "您好，\n\n"
    if events.len() > 0 {
        draft = draft + "感谢您的邮件。我已收到以下日程信息：\n\n"
        for e in events {
            draft = draft + "- 主题：" + e.title + "\n"
            if e.date != "" { draft = draft + "  日期：" + e.date + "\n" }
            if e.time != "" { draft = draft + "  时间：" + e.time + "\n" }
            if e.location != "" { draft = draft + "  地点：" + e.location + "\n" }
        }
        draft = draft + "我会准时参加。\n"
    }
    return draft
}
```
